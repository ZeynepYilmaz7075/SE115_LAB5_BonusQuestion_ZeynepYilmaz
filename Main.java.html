import java.util.Scanner; public class Main { public static void main(String[] args) { Scanner sc = new Scanner(System.in); System.out.println("Enter a positive integer: "); int n = sc.nextInt(); calculateFibonacci(n); } public static void calculateFibonacci(int n) { if (n < 0) { System.out.println("Invalid input"); return; } if (n == 0) { System.out.println("Fib(0)=0"); return; } if (n == 1) { System.out.println("Fib(1)=1"); } long [] fibArray = new long[n + 1]; fibArray[0] = 0; fibArray[1] = 1; for (int i = 2; i <= n; i++) { fibArray[i] = fibArray[i - 1] + fibArray[i - 2]; System.out.println("Fib("+i+") = " + fibArray[i]); } } }
